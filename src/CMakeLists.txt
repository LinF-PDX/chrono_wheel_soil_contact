cmake_minimum_required(VERSION 3.18)
project(chrono_single_wheel LANGUAGES CXX)

option(WHEEL_GUI "Show the wheel and rut using Irrlicht" ON)
if(WHEEL_GUI)
    find_package(Chrono 10.0 CONFIG QUIET COMPONENTS Vehicle Irrlicht)
else()
    find_package(Chrono 10.0 CONFIG QUIET COMPONENTS Vehicle)
endif()

# Chrono 10 checks for its model libraries even when using only SCM. Permit a
# partial SDK build if the only absent libraries are these unused extras.
if(NOT Chrono_FOUND)
    set(missing_required ${CHRONO_MISSING_LIBS_NDEBUG})
    list(REMOVE_ITEM missing_required ChronoModels_robot ChronoModels_vehicle Chrono_vehicle_irrlicht)
    if(NOT DEFINED CHRONO_MISSING_LIBS_NDEBUG OR missing_required)
        message(FATAL_ERROR "Build Chrono 10 with Vehicle (and Irrlicht for WHEEL_GUI), then set Chrono_DIR.")
    endif()
    set(required_targets Chrono_core Chrono_vehicle)
    if(WHEEL_GUI)
        list(APPEND required_targets Chrono_irrlicht)
    endif()
    foreach(library IN LISTS required_targets)
        if(NOT TARGET Chrono::${library})
            message(FATAL_ERROR "Required Chrono target is unavailable: ${library}")
        endif()
        get_target_property(location Chrono::${library} IMPORTED_LOCATION_RELEASE)
        if(NOT location)
            get_target_property(location Chrono::${library} IMPORTED_LOCATION)
        endif()
        if(NOT EXISTS "${location}")
            message(FATAL_ERROR "Build the required Chrono library: ${library}")
        endif()
    endforeach()
    message(STATUS "Using the built Chrono libraries required by the single-wheel simulation.")
endif()

add_executable(single_wheel main.cpp)
target_compile_features(single_wheel PRIVATE cxx_std_17)
target_compile_definitions(single_wheel PRIVATE "CHRONO_DATA_DIR=\"${CHRONO_DATA_DIR}\"")
# No vehicle model or vehicle-specific visualization is needed for one body.
target_link_libraries(single_wheel PRIVATE Chrono::Chrono_vehicle)
if(WHEEL_GUI)
    target_link_libraries(single_wheel PRIVATE Chrono::Chrono_irrlicht)
    target_compile_definitions(single_wheel PRIVATE WHEEL_GUI)
endif()
